    ```typescript
    import { describe, it, expect } from 'vitest';
    import { readFileSync } from 'fs';
    import { join } from 'path';
    import OpenAI from 'openai';
    import { verifyGarmentTypeMatch } from '../../src/lib/ai-image-generator.js';
    import labels from '../fixtures/garment-type/labels.json' with { type: 'json' };

    const FIXTURE_DIR = join(__dirname, '../fixtures/garment-type');

    type FixtureLabel = {
      expected_category: string;
      front_path: string;
      back_path: string;
      side_path: string;
      expected_match: { back: boolean; side: boolean };
    };

    describe.skipIf(!process.env.OPENAI_API_KEY)(
      'verifyGarmentTypeMatch — fixture set (real gpt-4o-mini, gated on OPENAI_API_KEY)',
      () => {
        const client = new OpenAI({
          apiKey: process.env.OPENAI_API_KEY,
          timeout: 60_000,
        });

        for (const [pid, rawLabel] of Object.entries(labels)) {
          const label = rawLabel as FixtureLabel;

          it(
            `${pid} (${label.expected_category}): back vs front → match=${label.expected_match.back}`,
            async () => {
              const frontBuf = readFileSync(join(FIXTURE_DIR, label.front_path));
              const backBuf = readFileSync(join(FIXTURE_DIR, label.back_path));
              const result = await verifyGarmentTypeMatch(client, backBuf, frontBuf);
              expect(result.match, `Expected match=${label.expected_match.back}, got reason: ${result.reason}`).toBe(label.expected_match.back);
            },
            30000,
          );

          it(
            `${pid} (${label.expected_category}): side vs front → match=${label.expected_match.side}`,
            async () => {
              const frontBuf = readFileSync(join(FIXTURE_DIR, label.front_path));
              const sideBuf = readFileSync(join(FIXTURE_DIR, label.side_path));
              const result = await verifyGarmentTypeMatch(client, sideBuf, frontBuf);
              expect(result.match, `Expected match=${label.expected_match.side}, got reason: ${result.reason}`).toBe(label.expected_match.side);
            },
            30000,
          );
        }
      },
    );
    ```

    Two import notes:
    - The `with { type: 'json' }` import attribute is Node.js standard (>=20.6 / >=22) for JSON imports in ESM. Verify the project's Node version supports this — `cat .nvmrc 2>/dev/null` or `node --version`. If unsupported, fall back to `import { readFileSync } from 'fs'` and `const labels = JSON.parse(readFileSync(join(FIXTURE_DIR, 'labels.json'), 'utf-8'))` at module scope.
    - `__dirname` works in CommonJS contexts. If the project is ESM-only and `__dirname` is undefined, replace with `const __dirname = dirname(fileURLToPath(import.meta.url));` after importing `dirname` from `'path'` and `fileURLToPath` from `'url'`. Verify by checking `tsconfig.json` `module` setting and how `tests/lib/prompt-templates.test.ts` handles paths.
